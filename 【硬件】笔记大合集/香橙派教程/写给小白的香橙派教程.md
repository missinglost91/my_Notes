```python
import cv2
import time
from rknnlite.api import RKNNLite
import numpy as np

# =================配置区 =================
# 【修改这里】换成你刚才转好的 rknn 模型名字
RKNN_MODEL = 'my.rknn' 
IMG_SIZE = 640  # 必须和你导出 ONNX 时设置的尺寸一模一样
# ==========================================

def main():
    print("⏳ [1/4] 正在实例化 NPU...")
    rknn = RKNNLite()
 
    print(f"⏳ [2/4] 正在加载模型: {RKNN_MODEL}")
    ret = rknn.load_rknn(RKNN_MODEL)
    if ret != 0:
        print("❌ 模型加载失败！请检查文件名对不对。")
        return

    print("⏳ [3/4] 正在分配 NPU 核心...")
    ret = rknn.init_runtime(core_mask=RKNNLite.NPU_CORE_0)
    if ret != 0:
        print("❌ NPU 运行时初始化失败！")
        return
    print("✅ NPU 启动成功！\n")

    print("⏳ [4/4] 正在打开摄像头...")
    cap = cv2.VideoCapture(0)
    cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)
 
    if not cap.isOpened():
        print("❌ 摄像头打开失败！")
        rknn.release()
        return

    print("🚀 测速开始！(按 Ctrl+C 停止)\n")
 
    try:
        while True:
            start_time = time.time()
 
            # 1. 读摄像头
            ret, frame = cap.read()
            if not ret:
                break

            # 2. 图像预处理 (缩放尺寸 + BGR转RGB)
            img = cv2.resize(frame, (IMG_SIZE, IMG_SIZE))
            img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
            img =img.reshape(1,IMG_SIZE,IMG_SIZE,3)
            # 3. NPU 硬件推理！
            outputs = rknn.inference(inputs=[img])

            # 4. 计算 FPS
            fps = 1.0 / (time.time() - start_time)
 
            # 打印测试信息：展示 NPU 吐出的原始张量形状和当前帧率
            # outputs[0].shape 通常会是类似 (1, 84, 8400) 的形状
            print(f"⚡ NPU FPS: {fps:.1f} | 成功吐出数据，输出层维度: {outputs[0].shape}", end='\r')

    except KeyboardInterrupt:
        print("\n\n🛑 测试由用户手动停止。")
    finally:
        cap.release()
        rknn.release()
        print("♻️ NPU 资源已安全释放。")

if __name__ == '__main__':
    main()
```


# 串口通信
## 香橙派部分代码

```python
import cv2
import time
import struct
import serial
import numpy as np
from rknnlite.api import RKNNLite

# ================= 配置区 =================
# 视觉配置
RKNN_MODEL = 'my.rknn'
IMG_SIZE = 640  # 请确保这里和你导出 ONNX 时的尺寸严格一致（分类模型通常是 224，按你实际情况来）

# 串口配置
SERIAL_PORT = '/dev/ttyS1'  
BAUD_RATE = 115200          
# ==========================================

def main():
    print("⏳ [1/5] 正在实例化 NPU...")
    rknn = RKNNLite()
    
    print(f"⏳ [2/5] 正在加载模型: {RKNN_MODEL}")
    ret = rknn.load_rknn(RKNN_MODEL)
    if ret != 0:
        print("❌ 模型加载失败！请检查文件名对不对。")
        return
        
    print("⏳ [3/5] 正在分配 NPU 核心...")
    ret = rknn.init_runtime(core_mask=RKNNLite.NPU_CORE_0)
    if ret != 0:
        print("❌ NPU 运行时初始化失败！")
        return
    print("✅ NPU 启动成功！")

    print("⏳ [4/5] 正在打开串口...")
    try:
        ser = serial.Serial(SERIAL_PORT, BAUD_RATE, timeout=0.1)
        print(f"✅ 串口开启成功！端口: {SERIAL_PORT} | 波特率: {BAUD_RATE}")
    except Exception as e:
        print(f"❌ 串口打开失败！请检查端口。\n错误信息: {e}")
        rknn.release()
        return

    print("⏳ [5/5] 正在打开摄像头...")
    cap = cv2.VideoCapture(0)
    cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)
    
    if not cap.isOpened():
        print("❌ 摄像头打开失败！")
        rknn.release()
        if 'ser' in locals() and ser.is_open:
            ser.close()
        return

    print("🚀 系统完美就绪，识别与传输开始！(按 Ctrl+C 停止)\n")

    try:
        while True:
            start_time = time.time()
            
            # 1. 读摄像头
            ret, frame = cap.read()
            if not ret:
                continue
                
            # 2. 图像预处理
            img = cv2.resize(frame, (IMG_SIZE, IMG_SIZE))
            img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
            img = img.reshape(1, IMG_SIZE, IMG_SIZE, 3)
            
            # 3. NPU 硬件推理
            outputs = rknn.inference(inputs=[img])
            
            # 4. 解析分类结果
            if outputs is not None:
                # 提取出概率数组，用 argmax 找出概率最高的索引（即手写数字 0-9）
                raw_tensor = outputs[0]
                detected_digit = int(np.argmax(raw_tensor))
            else:
                detected_digit = -1 # 未识别到或者异常
            
            # 5. 串口打包发送
            # 格式：帧头(0xAA) + 识别的数字(2字节) + Y位补0(2字节) + 帧尾(0x55)
            send_val_y = 0 
            send_frame = struct.pack('<BhhB', 0xAA, detected_digit, send_val_y, 0x55)
            ser.write(send_frame)

            # 6. 计算 FPS 并打印
            fps = 1.0 / (time.time() - start_time)
            print(f"⚡ FPS: {fps:.1f} | 👁️ 识别结果: {detected_digit} | 📤 发送 HEX: {send_frame.hex().upper()}     ", end='\r')

    except KeyboardInterrupt:
        print("\n\n🛑 测试由用户手动停止。")
    finally:
        cap.release()
        rknn.release()
        if 'ser' in locals() and ser.is_open:
            ser.close()
        print("♻️ 所有资源已安全释放。")

if __name__ == '__main__':
    main()
```

### 3. 编写串口中断回调函数（底层逻辑）

在 `main.c` 的最下方 `/* USER CODE BEGIN 4 */` 区域，编写解包状态机代码：



```C
/* USER CODE BEGIN 4 */
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    if (huart->Instance == USART1)
    {
        // 状态机解析帧
        if (rx_index == 0) 
        {
            // 等待帧头 0xAA
            if (rx_data == 0xAA) 
            {
                rx_buffer[rx_index++] = rx_data;
            }
        }
        else if (rx_index < 5) 
        {
            // 接收中间的数据位
            rx_buffer[rx_index++] = rx_data;
        }
        else if (rx_index == 5) 
        {
            // 接收第6个字节，并判断是否是帧尾 0x55
            rx_buffer[rx_index] = rx_data;
            if (rx_buffer[5] == 0x55) 
            {
                // 校验通过！按照小端序拼接 16 位整数
                vision_target_num = (int16_t)(rx_buffer[1] | (rx_buffer[2] << 8));
                
                // 提示主循环有新数据
                vision_update_flag = 1; 
            }
            // 一帧接收完毕（无论对错），索引清零，准备迎接下一帧
            rx_index = 0;
        }

        // 处理完毕后，务必再次开启中断
        HAL_UART_Receive_IT(&huart1, &rx_data, 1);
    }
}

// 极其重要：防止因为数据发送过快导致串口溢出死锁（ORE错误）
void HAL_UART_ErrorCallback(UART_HandleTypeDef *huart)
{
    if (huart->Instance == USART1)
    {
        __HAL_UART_CLEAR_OREFLAG(huart);
        HAL_UART_Receive_IT(&huart1, &rx_data, 1);
    }
}
/* USER CODE END 4 */
```
